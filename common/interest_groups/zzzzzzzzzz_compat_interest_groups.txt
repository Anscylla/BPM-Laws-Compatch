# --- ig_petty_bourgeoisie  (zrodlo: BPM/common/interest_groups/zzzz_petty_bourgeoisie.txt) ---
INJECT:ig_petty_bourgeoisie = {
	pop_weight = {
		value = 0
	
		if = {
			limit = { is_pop_type = academics }
			add = {
				desc = "POP_ACADEMICS"
				value = 50
			}
		}
		else_if = {
			limit = { is_pop_type = clerks }
			add = {
				desc = "POP_CLERKS"
				value = 100
			}
		}
		else_if = {
			limit = { is_pop_type = bureaucrats }
			add = {
				desc = "POP_BUREAUCRATS"
				value = 250
			}
		}
		else_if = {
			limit = { is_pop_type = capitalists }
			add = {
				desc = "POP_CAPITALISTS"
				value = 50
			}
		}
		else_if = {
			limit = { is_pop_type = officers }
			add = {
				desc = "POP_OFFICERS"
				value = 20
			}
		}
		else_if = {
			limit = {
				is_pop_type = aristocrats
				owner = { country_has_state_religion = rel:confucian }
			}
			add = {
				desc = "POP_ARISTOCRATS_CON"
				value = 250
			}
		}			
		else_if = {
			limit = { is_pop_type = aristocrats }
			add = {
				desc = "POP_ARISTOCRATS"
				value = 50
			}
		}
				
		add = {
			desc = "MILD_DISCRIMINATION"
			if = {
				limit = {
					pop_acceptance >= acceptance_status_3
					pop_acceptance <= acceptance_status_4
					bpm_pop_employment_is_rural = no
				}
				value = 100
			}
		}

		if = {
			limit = {
				strata = middle
				owner = {
					OR = {
						has_law = law_type:law_landed_voting
						has_law = law_type:law_wealth_voting
						has_law = law_type:law_census_voting
						has_law = law_type:law_appointed_bureaucrats
						has_law_or_variant = law_type:law_meritocratic_bureaucracy	# Laws+
					}
				}
			}
			add = {
				desc = "POLITICAL_MACHINES"
				value = 100
			}
		}
		else_if = {
			limit = {
				strata = lower
				owner = {
					OR = {
						has_law = law_type:law_census_voting
						has_law = law_type:law_elected_bureaucrats
					}
				}
			}
			add = {
				desc = "POLITICAL_MACHINES"
				value = 50
			}
		}

		if = {
			limit = {
				owner = {
					c:BIC ?= this
				}
				#OR = {
				#	is_pop_type = bureaucrats
				#	is_pop_type = clerks
				#}
			}
			if = {
				limit = {
					culture = {
						has_discrimination_trait_group = heritage_group_european
					}
				}
				multiply = {
					desc = "POP_INDIA_COLONIAL_RULE"
					value = 0.01 #0.25
				}
			}
			else = {
				multiply = {
					desc = "POP_INDIA_CIVIL_SERVICE"
					value = 0.3 #1.5
				}
			}
		}
		
		add = {
			desc = "POP_MILITARY_BUREAUCRACY"
			if = {
				limit = {
					is_pop_type = bureaucrats
					owner = {
						c:ALK ?= this
						has_government_type = gov_chartered_company
					}
				}
				multiply = 0.25
			}
		}

		multiply = ig_attraction_to_interests_from_stability
	}
}

# --- ig_devout  (zrodlo: BPM/common/interest_groups/zzzz_devout.txt) ---
INJECT:ig_devout = {
	pop_weight = {
		value = 25

		if = {
			limit = { is_pop_type = peasants }
			add = {
				desc = "POP_PEASANTS"
				value = 100
			}
		}
		else_if = {
			limit = { is_pop_type = clergymen }
			add = {
				desc = "POP_CLERGYMEN"
				value = 500
			}
		}
		
		if = {
			limit = {
				owner = { has_law_or_variant = law_type:law_state_religion }
			}
			multiply = {
				desc = "STATE_RELIGION_LAW"
				value = 2.0
			}
		}
		else_if = {
			limit = {
				owner = { has_law_or_variant = law_type:law_total_separation }
			}
			multiply = {
				desc = "TOTAL_SEPARATION_LAW"
				value = 0.8
			}
		}
		else_if = {
			limit = {
				owner = { has_law = law_type:law_state_atheism }
			}
			multiply = {
				desc = "STATE_ATHEISM_LAW"
				value = 0.1
			}
		}

		add = {
			desc = "RELIGIOUS_INSTITUTIONS"
			if = {
				limit = {
					owner = {
						OR = {
							has_law = law_type:law_religious_schools
							has_law = law_type:law_charitable_health_system
						}
					}
				}
				value = 50
			}
		}

		add = {
			desc = "DEVOUT_LITERACY"
			if = {
				limit = { 
					NOT = { 
						is_pop_type = clergymen
					}
				}
				value = literacy_rate
				multiply = -100
				add = 50
				max = 0
			}
		}

		add = {
			desc = "THEOCRACY"
			if = {
				limit = {
					owner = { has_law = law_type:law_theocracy }
					has_law = law_type:law_gwageo	# Laws+
				}
				value = 100
			}
		}

		multiply = ig_attraction_to_interests_from_stability	
		
		multiply = {
			desc = "RELIGION_SUPPRESSED"
			value = 1.0
			if = {
				limit = {
					is_pop_type = clergymen
					owner = {
						has_law_or_variant = law_type:law_total_separation
					}
				}
				value = 1.5
			}
		}		
	}
}
