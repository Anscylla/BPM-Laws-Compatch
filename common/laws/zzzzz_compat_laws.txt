# --- law_autocracy  (zrodlo: BPM/BPM_distribution_of_power.txt) ---
REPLACE:law_autocracy = {	
	group = lawgroup_distribution_of_power
	icon = "gfx/interface/icons/law_icons/autocracy.dds"
	
	progressiveness = -100

	on_activate = {
		bpm_disable_elective_laws = yes
		if = {
			limit = { country_has_voting_franchise = no }
			cleanup_voting_strength_modifiers = yes
			bpm_remove_voting_strength_modifiers = yes
			if = {
				limit = { bpm_country_is_republic = yes }
				scripted_effect_parties_emergence = yes
			}
			else = {
				scripted_effect_parties_disappearence = yes
			}
		}
		else = {
			random_law = {
				limit = {
					type = law_type:law_autocracy
				}
				save_scope_as = relevant_voting_law
			}
			# if = {
			# 	limit = {
			# 		game_date > 1836.1.1
			# 	}
			# 	initialize_electoral_confidence = yes
			# }
			handle_possible_absent_caciquismo_variables = yes
			inherit_caciquismo_effect = yes
			inherit_free_elections_effect = yes
			scripted_effect_parties_emergence = yes
		}
		if = {
			limit = { 
				bpm_country_is_republic = yes	
				NOT = { has_modifier = bpm_republic_gov_size }
			}
			add_modifier = bpm_republic_gov_size
		}
		if = { # Laws+ compat: ustroj bez legislatywy wylacza wybory
			limit = { NOT = { has_law = law_type:law_no_election } }
			activate_law = law_type:law_no_election
		}
	}

	modifier = {
		country_legitimacy_govt_total_clout_add = 100
		country_legitimacy_govt_size_add = 2
		country_legitimacy_ideological_incoherence_mult = 0.25
		country_bureaucrats_pol_str_mult = 0.5
		country_peasants_pol_str_mult = -0.4
		state_pop_pol_str_mult = -0.5
		state_pop_pol_str_add = -2
		country_electoral_confidence_impact_mult = -1
		country_authority_add = 200
		# country_bpm_base_mandate_add = 60
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_distribution_of_power_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0
		
		if = {
			limit = {
				has_law = law_type:law_autocracy
				OR = {
					AND = {
						power_bloc ?= { has_identity = identity:identity_ideological_union }
						has_strategy = ai_strategy_conservative_agenda					
					}
					has_strategy = ai_strategy_reactionary_agenda
				}
				scope:target_country = { 
					NOR = { # Is their law different enough from what we want to impose?
						has_law = law_type:law_oligarchy
						has_law = law_type:law_landed_voting
					}
				}
			}
			
			add = base_impose_law_weight
		}
	}
}

# --- law_census_voting  (zrodlo: BPM/BPM_distribution_of_power.txt) ---
REPLACE:law_census_voting = {
	group = lawgroup_distribution_of_power
	
	icon = "gfx/interface/icons/law_icons/census_voting.dds"
	
	progressiveness = 50

	unlocking_technologies = {
		democracy
	}

	disallowing_laws = {
		law_chiefdom
	}

	modifier = {
		country_voting_power_wealth_threshold_add = 10
		country_voting_power_from_literacy_add = 25
		country_authority_add = 100
		state_pop_pol_str_add = -0.5

		country_electoral_confidence_impact_mult = -0.1

		country_legitimacy_govt_total_votes_add = 100
		country_legitimacy_govt_total_clout_add = 75

		state_radicals_from_political_movements_mult = -0.05
		country_rigidity_baseline_add = -15
		country_bpm_labor_consciousness_target_mult = -0.05
		
	}

	on_enact = {
		if = {
			limit = {has_law = law_type:law_bakufu}
			custom_tooltip = bpm_voting_end_bakufu
		}
	}

	on_activate = {
		scripted_effect_parties_emergence = yes
		if = {
			limit = { has_modifier = bpm_republic_gov_size }
			remove_modifier = bpm_republic_gov_size
		}
		if = {
			limit = {
				game_date > 1836.1.1
			}
			initialize_electoral_confidence = yes
		}
		random_law = {
			limit = {
				type = law_type:law_census_voting
			}
			save_scope_as = relevant_voting_law
		}
		handle_possible_absent_caciquismo_variables = yes
		inherit_caciquismo_effect = yes
		inherit_free_elections_effect = yes
		if = {
			limit = {
				has_law = law_type:law_monarchy
				has_variable = king_in_parliament_possible
				NOT = {
					has_journal_entry = je_king_in_parliament
				}
			}
			custom_tooltip = {
				text = king_in_parliament_tt
				add_journal_entry = { type = je_king_in_parliament }
			}
		}
		if = { # Laws+ compat: przywroc glosowanie po wyjsciu z braku wyborow
			limit = { has_law = law_type:law_no_election }
			if = {
				limit = { country_has_monarchy_law = yes }
				activate_law = law_type:law_semi_constitutional
			}
			else = { activate_law = law_type:law_non_secret_ballot }
		}
	}
	
	can_impose = {
		country_has_voting_franchise = yes # modification of the standard trigger, only countries with a voting franchise can impose democracy on others
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_distribution_of_power_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0
		
		if = {
			limit = {
				has_law = law_type:law_census_voting
				has_strategy = ai_strategy_progressive_agenda			
				scope:target_country = { # Is their law different enough from what we want to impose?
					NOR = { 
						has_law = law_type:law_wealth_voting 
						has_law_or_variant = law_type:law_weighted_universal_voting	# Laws+
						has_law = law_type:law_universal_suffrage 
					}
				}
			}
			
			add = base_impose_law_weight
		}
	}
}
