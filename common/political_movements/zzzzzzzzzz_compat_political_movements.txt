# --- movement_fascist  (zrodlo: BPM/common/political_movements/zzzz_bpm_ideological_reactionary_movements.txt) ---
INJECT:movement_fascist = {
	character_ideologies = {
		ideology_fascist
		ideology_integralist
		ideology_corporatist_leader
		ideology_clerical_fascist	# Laws+
		ideology_national_socialist	# Laws+
	}
	character_support_trigger = {
		OR = {
			bpm_ig_is_fascist = yes
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_clerical_fascist	# Laws+
			has_ideology = ideology:ideology_national_socialist	# Laws+
			has_ideology = ideology:ideology_integralist
			has_ideology = ideology:ideology_corporatist_leader
			has_ideology = ideology:ideology_ethno_nationalist
		}
	}
	character_support_weight = {
		if = {
			limit = {
				OR = {
					bpm_ig_is_fascist = yes
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_clerical_fascist	# Laws+
					has_ideology = ideology:ideology_national_socialist	# Laws+
					has_ideology = ideology:ideology_integralist
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_corporatist_leader
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
			multiply = {
				value = 2
			}
		}
	}
}

# --- movement_cultural_majority  (zrodlo: BPM/common/political_movements/zzzz_bpm_cultural_movement.txt) ---
INJECT:movement_cultural_majority = {
	character_support_trigger = {
		culture = {
			is_primary_culture_of = root.owner
		}
		OR = {
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_jingoist_leader
			has_ideology = ideology:ideology_ethno_nationalist
			has_ideology = ideology:ideology_clerical_fascist	# Laws+
			has_ideology = ideology:ideology_national_socialist	# Laws+
			bpm_character_is_modern_nationalist = yes
			bpm_character_is_traditional_nationalist = yes
			law_stance = {
				law = law_type:law_ethnostate
				value > neutral
			}
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ethno_nationalist
					has_ideology = ideology:ideology_clerical_fascist	# Laws+
					has_ideology = ideology:ideology_national_socialist	# Laws+
					law_stance = {
						law = law_type:law_ethnostate
						value > neutral
					}
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					bpm_character_is_traditional_nationalist = yes
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_jingoist_leader
				}
			}
			multiply = {
				value = 2
			}
		}
		else_if = {
			limit = {
				OR = {
					bpm_character_is_cosmopolitan = yes
					has_progressive_ideology = yes
					has_socialist_ideology = yes
				}
			}
			multiply = 0
		}
	}
}

# --- movement_carlist  (zrodlo: BPM/common/political_movements/zzzz_bpm_country_specific_ideological_movements.txt) ---
INJECT:movement_carlist = {
	character_support_trigger = {
		OR = {
			character_has_carlist_ideology = yes
			has_ideology = ideology:ideology_clerical_fascist	# Laws+
			has_ideology = ideology:ideology_traditionalist
			has_ideology = ideology:ideology_authoritarian
			has_ideology = ideology:ideology_luddite
			has_ideology = ideology:ideology_reactionary_anti_capitalist
			interest_group ?= {
				OR = {
					is_interest_group_type = ig_reactionaries
					has_ideology = ideology:ideology_reactionary
					ig_has_carlist_ideology = yes
					has_ideology = ideology:ideology_patriarchal
				}
			}
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				character_has_carlist_ideology = yes
				has_ideology = ideology:ideology_clerical_fascist	# Laws+
			}
			multiply = {
				value = 5
			}
		}
		if = {
			limit = {
				scope:interest_group ?= {
					is_interest_group_type = ig_devout
				}
			}
			multiply = {
				value = 2
			}
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_protectionist
					has_ideology = ideology:ideology_traditionalist
					has_ideology = ideology:ideology_luddite
					has_ideology = ideology:ideology_reactionary_anti_capitalist
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				AND = {
					has_ideology = ideology:ideology_authoritarian
					interest_group ?= {
						OR = {
							is_interest_group_type = ig_intelligentsia 
							is_interest_group_type = ig_trade_unions 
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
		if = {
			limit = {
				interest_group ?= {
					OR = {
						has_ideology = ideology:ideology_reactionary
						has_ideology = ideology:ideology_patriarchal
					}
				}
			}
			multiply = {
				value = 2
			}
		}
		
	}
	additional_radicalism_factors = {
		add = { value = bpm_additional_radicalism_factors desc = "bpm_additional_radicalism_factors" }
		owner ?= { if = {
				limit = {		
					owner = {has_law = law_type:law_monarchy}
				}
				
				if = {
					limit = {
						owner = {
							OR = {
								is_enacting_law = law_type:law_presidential_republic
								is_enacting_law = law_type:law_parliamentary_republic
								is_enacting_law = law_type:law_corporate_state
								is_enacting_law = law_type:law_theocracy
								is_enacting_law = law_type:law_council_republic	
							}
						}			
					}
					
					add = {
						value = 0.25
						desc = "MONARCHY_BEING_ABOLISHED"
					}					
				}
				# else = {
				# 	add = { # Only care about legitimacy in the context of a monarchy that isn't under direct threat
				# 		value = 50
				# 		subtract = legitimacy
				# 		multiply = 0.005
				# 		desc = "concept_legitimacy"
				# 	}				
				# }
			} }
		if = {
			limit = {
				owner ?= {
					has_law = law_type:law_monarchy
					ruler ?= {
						NOT = {
							character_has_carlist_ideology = yes
							has_ideology = ideology:ideology_clerical_fascist	# Laws+
						}
					}
				}
			}
			add = {
				value = 0.2
				desc = "NON_CARLIST_MONARCH"
			}
		}
		if = {
			limit = {
				exists = c:PAP
				c:PAP ?= {
					country_has_state_religion = rel:catholic
					has_law_or_variant = law_type:law_theocracy
				}
				"owner.relations(c:PAP)" < 0
			}
			add = {
				desc = "POOR_RELATIONS_WITH_PAPACY"
				value = "owner.relations(c:PAP)"
				divide = 100
				multiply = -0.15
			}
		}
		if = {
			limit = {
				owner ?= {
					c:SPA ?= this
					has_journal_entry = je_the_first_carlist_war
				}
			}
			add = {
				value = -0.95
				desc = "INSURRECTIONARY_CARLISTS"
			}
		}
	}
}

# --- movement_modernizer  (zrodlo: BPM/common/political_movements/zzzz_bpm_ideological_liberal_movements.txt) ---
INJECT:movement_modernizer = {
	character_support_trigger = {
		bpm_character_is_traditional_econ = no
		OR = {
			has_ideology = ideology:ideology_reformer
			has_ideology = ideology:ideology_anarcho_liberal	# Laws+
			has_ideology = ideology:ideology_protectionist
			has_ideology = ideology:ideology_positivist
			has_ideology = ideology:ideology_abolitionist
			has_ideology = ideology:ideology_modernizer_leader
			has_ideology = ideology:ideology_sovereignist
			bpm_ig_is_liberal = yes
		}
	}
	character_support_weight = {
		value = 25
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_protectionist
					has_ideology = ideology:ideology_reformer
					has_ideology = ideology:ideology_anarcho_liberal	# Laws+
					has_ideology = ideology:ideology_sovereignist
					has_ideology = ideology:ideology_modernizer_leader
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_positivist
					has_ideology = ideology:ideology_abolitionist
				}
			}
			multiply = {
				value = 3
			}
		}
		if = {
			limit = {
				interest_group ?= {
					OR = {
						has_ideology = ideology:ideology_modernizer
						has_ideology = ideology:ideology_liberal
						has_ideology = ideology:ideology_liberal_modern
						has_ideology = ideology:ideology_plutocratic
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
}
